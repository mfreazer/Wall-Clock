cmake_minimum_required(VERSION 3.16)
project(wall_clock LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
foreach(test_name IN ITEMS engine requirements)
  add_executable(${test_name}_test test/${test_name}_test.cpp)
  target_include_directories(${test_name}_test PRIVATE include)
  target_compile_options(${test_name}_test PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME ${test_name} COMMAND ${test_name}_test)
endforeach()
find_program(NODE_EXECUTABLE node)
if(NODE_EXECUTABLE)
  add_test(NAME web_ui COMMAND ${NODE_EXECUTABLE} test/web_ui_test.js)
  set_tests_properties(web_ui PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
else()
  message(STATUS "Node.js not found: embedded web UI test will not be registered")
endif()
